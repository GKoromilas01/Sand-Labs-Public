import { createClient } from "npm:@supabase/supabase-js@2.49.8";
import { Chess } from "npm:chess.js@1.4.0";
(function(root){'use strict';const SIZE=9,KOMI=6.5,other=c=>3-c,hash=b=>b.join('');
const adjacent=i=>{const r=Math.floor(i/SIZE),c=i%SIZE;return [r>0?i-SIZE:-1,r<SIZE-1?i+SIZE:-1,c>0?i-1:-1,c<SIZE-1?i+1:-1].filter(x=>x>=0);};
function group(board,start){if(!board[start])return {stones:[],liberties:[]};const colour=board[start],stones=new Set([start]),liberties=new Set(),stack=[start];while(stack.length){for(const i of adjacent(stack.pop())){if(!board[i])liberties.add(i);else if(board[i]===colour&&!stones.has(i)){stones.add(i);stack.push(i);}}}return {stones:[...stones],liberties:[...liberties]};}
const initial=()=>({board:Array(81).fill(0),turn:1,captures:[0,0,0],passes:0,phase:'play',seen:['0'.repeat(81)],moves:[],dead:[],agreements:[]});
function play(s,point){if(s.phase!=='play')throw Error('Finish scoring or choose Play on first.');if(!Number.isInteger(point)||point<0||point>=81||s.board[point])throw Error('Choose an empty intersection.');const board=[...s.board],captures=[...s.captures];board[point]=s.turn;for(const i of adjacent(point)){if(board[i]!==other(s.turn))continue;const g=group(board,i);if(!g.liberties.length){for(const p of g.stones)board[p]=0;captures[s.turn]+=g.stones.length;}}if(!group(board,point).liberties.length)throw Error('Suicide is not allowed.');const h=hash(board);if(s.seen.includes(h))throw Error('Positional superko: this board position has already occurred.');return {...s,board,captures,turn:other(s.turn),passes:0,seen:[...s.seen,h],moves:[...s.moves,{kind:'play',point}],dead:[],agreements:[]};}
function pass(s){if(s.phase!=='play')throw Error('The game is in scoring.');const passes=s.passes+1;return {...s,turn:other(s.turn),passes,phase:passes>=2?'score':'play',moves:[...s.moves,{kind:'pass'}],dead:[],agreements:[]};}
function resume(s){if(s.phase!=='score')throw Error('Only an unagreed score can return to play.');return {...s,phase:'play',passes:0,dead:[],agreements:[],moves:[...s.moves,{kind:'resume'}]};}
function markDead(s,point){if(s.phase!=='score')throw Error('Mark dead stones only during scoring.');if(!s.board[point])return s;const stones=group(s.board,point).stones,dead=new Set(s.dead),remove=stones.every(x=>dead.has(x));for(const p of stones)if(remove)dead.delete(p);else dead.add(p);return {...s,dead:[...dead],agreements:[]};}
function score(s){const board=[...s.board];for(const p of s.dead)board[p]=0;const area=[0,0,0],visited=new Set();for(let i=0;i<81;i++){if(board[i]){area[board[i]]++;continue;}if(visited.has(i))continue;const empty=[],border=new Set(),stack=[i];visited.add(i);while(stack.length){const p=stack.pop();empty.push(p);for(const q of adjacent(p)){if(board[q])border.add(board[q]);else if(!visited.has(q)){visited.add(q);stack.push(q);}}}if(border.size===1)area[[...border][0]]+=empty.length;}return {black:area[1],white:area[2]+KOMI,komi:KOMI};}
function agree(s,colour){if(s.phase!=='score'||![1,2].includes(colour))throw Error('Scoring agreement is not available.');const agreements=[...new Set([...s.agreements,colour])];return {...s,agreements,phase:agreements.length===2?'finished':'score'};}
function replay(moves){if(!Array.isArray(moves)||moves.length>2000)throw Error('Invalid move record.');let s=initial();for(const m of moves){if(m?.kind==='play')s=play(s,m.point);else if(m?.kind==='pass')s=pass(s);else if(m?.kind==='resume')s=resume(s);else throw Error('Unknown move.');}return s;}
function parse(raw){if(typeof raw!=='string'||raw.length>200000)throw Error('Backup is too large.');const d=JSON.parse(raw);if(d.application!=='sandlabs-go'||d.version!==1||d.size!==9||d.komi!==KOMI)throw Error('Not a Go9x9 backup.');let s=replay(d.moves);if(!Array.isArray(d.dead)||!Array.isArray(d.agreements)||d.dead.length>81||new Set(d.dead).size!==d.dead.length||new Set(d.agreements).size!==d.agreements.length)throw Error('Invalid scoring record.');for(const p of d.dead)if(!Number.isInteger(p)||p<0||p>=81||!s.board[p]||s.phase!=='score')throw Error('Invalid dead group.');for(const p of d.dead)if(group(s.board,p).stones.some(x=>!d.dead.includes(x)))throw Error('Mark complete groups.');s={...s,dead:d.dead};for(const c of d.agreements)s=agree(s,c);return s;}
const snapshot=s=>({application:'sandlabs-go',version:1,size:9,komi:KOMI,moves:s.moves,dead:s.dead,agreements:s.agreements});
const api={SIZE,KOMI,initial,adjacent,group,play,pass,resume,markDead,score,agree,replay,parse,snapshot};if(typeof module!=='undefined')module.exports=api;root.GoRules=api;
})(globalThis);

const GoRules = (globalThis as any).GoRules;
const url=Deno.env.get('SUPABASE_URL')!, secret=Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!;
const admin=createClient(url,secret,{auth:{persistSession:false,autoRefreshToken:false}});
const origins=(Deno.env.get('SAND_ALLOWED_ORIGINS')||'').split(',').map(x=>x.trim()).filter(Boolean);
Deno.serve(async req=>{
 const origin=req.headers.get('origin')||'';
 const headers={'Content-Type':'application/json','Access-Control-Allow-Origin':origins.includes(origin)?origin:'null','Vary':'Origin','Access-Control-Allow-Headers':'authorization, apikey, content-type','Access-Control-Allow-Methods':'POST, OPTIONS'};
 const reply=(data:unknown,status=200)=>new Response(JSON.stringify(data),{status,headers});
 if(!origins.includes(origin))return reply({error:'Origin not allowed'},403);
 if(req.method==='OPTIONS')return new Response(null,{status:204,headers});
 if(req.method!=='POST')return reply({error:'Use POST'},405);
 try{
 const bearer=req.headers.get('authorization')?.match(/^Bearer (.+)$/i)?.[1];
 if(!bearer)return reply({error:'Sign in required'},401);
 const {data:auth,error:authError}=await admin.auth.getUser(bearer);
 if(authError||!auth.user)return reply({error:'Session expired; sign in again'},401);
 const actor=auth.user.id;
 const raw=await req.text();if(raw.length>4096)return reply({error:'Request too large'},413);
 const body=JSON.parse(raw);
 let row:any;
 if(body.op==='join'){
 if(!['chess','go'].includes(body.game))throw Error('Unknown game');
 const {data,error}=await admin.rpc('board_join',{actor,wanted:body.game,initial_position:body.game==='chess'?{pgn:''}:GoRules.snapshot(GoRules.initial()),initial_turn:body.game==='chess'?'w':'b'});
 if(error)throw Error(error.message);row=data;
 }else if(body.op==='current'){
 const {data,error}=await admin.from('board_matches').select('*').or(`white.eq.${actor},black.eq.${actor}`).in('status',['waiting','active']).order('updated_at',{ascending:false}).limit(1).maybeSingle();
 if(error)throw Error(error.message);row=data;
 }else if(body.op==='move'||body.op==='leave'){
 if(typeof body.id!=='string'||!Number.isInteger(body.revision))throw Error('Invalid match revision');
 const {data,error}=await admin.from('board_matches').select('*').eq('id',body.id).maybeSingle();
 if(error||!data||(data.white!==actor&&data.black!==actor))return reply({error:'Match unavailable'},403);
 row=data;const colour=row.white===actor?'w':'b';
 if(row.revision!==body.revision||row.status==='finished')return reply({error:'Stale revision; refresh match'},409);
 let position=row.position,turn=row.turn,status=row.status,result=row.result;
 if(body.op==='leave'){status='finished';result=row.status==='waiting'?'cancelled':(colour==='w'?'black wins by resignation':'white wins by resignation');}
 else{
 if(row.status!=='active')throw Error('Waiting for an opponent');
 const a=body.action;
 if(row.game==='chess'){
 if(colour!==turn)throw Error('Wait for your turn');
 if(!a||typeof a.from!=='string'||typeof a.to!=='string'||(a.promotion&&!['q','r','b','n'].includes(a.promotion)))throw Error('Invalid move');
 const chess=new Chess();chess.loadPgn(position.pgn);chess.move({from:a.from,to:a.to,promotion:a.promotion||'q'});
 position={pgn:chess.pgn()};turn=chess.turn();if(chess.isGameOver()){status='finished';result=chess.isCheckmate()?(turn==='w'?'black wins by checkmate':'white wins by checkmate'):'draw';}
 }else{
 let s=GoRules.parse(JSON.stringify(position));
 if(['play','pass'].includes(a?.kind)&&colour!==turn)throw Error('Wait for your turn');
 switch(a?.kind){case 'play':s=GoRules.play(s,a.point);break;case 'pass':s=GoRules.pass(s);break;case 'resume':s=GoRules.resume(s);break;case 'markDead':if(!Number.isInteger(a.point)||a.point<0||a.point>80)throw Error('Invalid point');s=GoRules.markDead(s,a.point);break;case 'agree':s=GoRules.agree(s,colour==='b'?1:2);break;default:throw Error('Unknown action');}
 position=GoRules.snapshot(s);turn=s.turn===1?'b':'w';if(s.phase==='finished'){status='finished';const score=GoRules.score(s);result=score.black>score.white?'black wins on area':'white wins on area';}
 }
 }
 const committed=await admin.rpc('board_commit',{actor,match_id:row.id,expected:body.revision,next_position:position,next_turn:turn,next_status:status,next_result:result});
 if(committed.error)return reply({error:'Match changed; refresh before trying again'},409);row=committed.data;
 }else throw Error('Unknown operation');
 return reply({match:row?{id:row.id,game:row.game,colour:row.white===actor?'w':'b',status:row.status,revision:row.revision,position:row.position,turn:row.turn,result:row.result}:null});
 }catch(e){return reply({error:e instanceof Error?e.message:'Request failed'},400);}
});
